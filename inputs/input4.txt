fun int comparar(int a, int b)
    return a <=> b
endfun

fun int main()
    var int* arr;
    var int i, j, temp, n;
    n = 5;
    arr = malloc(n * 8);
    arr[0] = 64;
    arr[1] = 34;
    arr[2] = 25;
    arr[3] = 12;
    arr[4] = 22;

    i = 0;
    while i < n - 1 do
        j = 0;
        while j < n - i - 1 do
            if comparar(arr[j], arr[j + 1]) == 1 then
                temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp
            endif;
            j = j + 1
        enddo;
        i = i + 1
    enddo;

    i = 0;
    while i < n do
        print(arr[i]);
        i = i + 1
    enddo;

    free(arr)
endfun
