fun int clamp(int v, int minV, int maxV)
    if (v <=> minV) == -1 then
        return minV
    else
        if (v <=> maxV) == 1 then
            return maxV
        else
            return v
        endif
    endif
endfun

fun int procesarPila(stack<int> s)
    var int prev, curr, diff, balance;
    balance = 0;
    if s.empty() then
        return 0
    endif;
    prev = s.pop();
    while s.empty() == false do
        curr = s.pop();
        diff = prev <=> curr;
        balance = balance + diff;
        prev = curr
    enddo;
    return balance
endfun

fun int main()
    var stack<int> s;
    var int res1, res2;
    s.push(10);
    s.push(20);
    s.push(30);
    s.push(40);
    res1 = procesarPila(s) ?: 999;
    print(res1);

fun int ranking(int* scores, int n)
    var int i, totalDiff;
    totalDiff = 0;
    i = 0;
    while i < n - 1 do
        totalDiff = totalDiff + (scores[i] <=> scores[i + 1]);
        i = i + 1
    enddo;
    return totalDiff
endfun
